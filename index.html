Supabase SQL Schema & Admin RLS (Run in Supabase SQL Editor)

Configures vehicles and bookings tables with explicit privileges for Admin User ID d3e8c025-a4f4-45f6-99d7-852bf305a92e.

-- 1. Create Vehicles Table (UUID Primary Key)
CREATE TABLE IF NOT EXISTS public.vehicles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  category TEXT NOT NULL,
  description TEXT,
  specs TEXT,
  image_url TEXT NOT NULL,
  rate_1hr NUMERIC NOT NULL,
  rate_3hr NUMERIC NOT NULL,
  rate_6hr NUMERIC NOT NULL,
  rate_12hr NUMERIC NOT NULL,
  rate_daily NUMERIC NOT NULL,
  is_active BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Create Bookings Table with overlap checking support
CREATE TABLE IF NOT EXISTS public.bookings (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  vehicle_id UUID REFERENCES public.vehicles(id) ON DELETE CASCADE,
  vehicle_name TEXT NOT NULL,
  customer_name TEXT NOT NULL,
  customer_phone TEXT NOT NULL,
  pickup_datetime TEXT NOT NULL,
  dropoff_datetime TEXT NOT NULL,
  service_type TEXT DEFAULT 'Self-Drive Excursion',
  delivery_location TEXT,
  estimated_total NUMERIC NOT NULL,
  status TEXT DEFAULT 'confirmed',
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. Row Level Security (RLS) & Admin Access for d3e8c025-a4f4-45f6-99d7-852bf305a92e
